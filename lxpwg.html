Admin tool for managing LearnX access credentials

Not connected - Configure Supabase first

📚 Quick Setup Guide:

  1. Go to Supabase Dashboard → SQL Editor
  2. Run this SQL command to create the table:
CREATE TABLE IF NOT EXISTS public.credentials (
  id BIGSERIAL PRIMARY KEY,
  access_code TEXT UNIQUE NOT NULL,
  password_hash TEXT NOT NULL,
  created_at TIMESTAMPTZ DEFAULT NOW(),
  expiry_days INTEGER DEFAULT 365,
  last_login TIMESTAMPTZ,
  is_active BOOLEAN DEFAULT TRUE
);

ALTER TABLE public.credentials ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Allow all" ON public.credentials FOR ALL TO anon USING (true) WITH CHECK (true);

GRANT ALL ON public.credentials TO anon;
GRANT USAGE, SELECT ON SEQUENCE credentials_id_seq TO anon;
